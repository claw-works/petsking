帮我写个带重试的异步 fetch 封装，最多 3 次，指数退避

好的，我来实现一个带指数退避重试的 fetchWithRetry。整体思路：

  • 失败（网络错误或 5xx）才重试，4xx 直接抛出
  • 退避间隔 base * 2^n，并加少量随机抖动避免雪崩
  • 超过最大次数后抛出最后一次错误
async function fetchWithRetry(url, opts = {}, max = 3, base = 500) {
  let lastErr;
  for (let n = 0; n <= max; n++) {
    try {
      const res = await fetch(url, opts);
      if (res.status < 500) return res;   // 4xx 不重试
      throw new Error('HTTP ' + res.status);
    } catch (err) {
      lastErr = err;
      if (n === max) break;
      const wait = base * 2 ** n + Math.random() * 100;
      await new Promise(r => setTimeout(r, wait));
    }
  }
  throw lastErr;
}
